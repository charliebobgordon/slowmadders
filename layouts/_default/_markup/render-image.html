{{- $image := .Page.Resources.GetMatch .Destination -}} {{- $alt := .PlainText -}} {{- $title := .Title -}} {{- if not $image -}} {{- if not (or (strings.HasPrefix .Destination "http") (strings.HasPrefix .Destination "/")) -}} {{- warnf "Unable to find '%s' -- ensure image exists alongside document in page bundle" .Destination -}} {{- end -}} {{- else if strings.HasSuffix $image.Name ".gif" -}} {{- else -}} {{- $origWidth := $image.Width -}} {{- $srcset := slice -}} {{- range slice 480 728 -}} {{- if lt (mul . 1.2) $origWidth -}} {{- $thumb := $image.Resize (printf "%dx webp q90" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $thumb.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $full := $image.Resize (printf "%dx webp q90" $origWidth) -}} {{- $srcset = $srcset | append (printf "%s %dw" $full.RelPermalink $origWidth) -}} {{- end -}}
